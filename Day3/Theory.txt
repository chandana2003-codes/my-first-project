Topics:
SELECT, WHERE, ORDER BY, LIMIT, TOP,

AGGREGATE FUNCTIONS, LOGICAL OPERATORS

1. SELECT + WHERE - Filter cheyadam

WHERE = condition petti data filter cheyadam

SELECT * FROM employees WHERE salary > 10000;
SELECT * FROM employees WHERE department_id = 9;

LOGICAL OPERATORS: AND, OR, NOT
AND = rendu conditions true avvali
OR = okati true ayina chalu
NOT = opposite

Example:
SELECT * FROM employees WHERE salary > 5000 AND department_id = 6;
SELECT * FROM employees WHERE department_id = 9 OR department_id = 10;
SELECT * FROM employees WHERE NOT department_id = 9;

2. ORDER BY - Sort cheyadam

ORDER BY = data ni sort cheyadam (small to big / big to small)

SELECT * FROM employees ORDER BY salary ASC;  -- ASC = low to high (default)
SELECT * FROM employees ORDER BY salary DESC; -- DESC = high to low
SELECT * FROM employees ORDER BY hire_date DESC;

3. LIMIT / TOP - Konni rows matrame

Snowflake lo LIMIT, SQL Server lo TOP

SELECT * FROM employees LIMIT 10;      -- First 10 rows
SELECT TOP 5 * FROM employees;         -- First 5 rows
SELECT * FROM employees ORDER BY salary DESC LIMIT 3; -- Top 3 highest salary

4. AGGREGATE FUNCTIONS - Lekkalu veyadam

COUNT() - entha mandi unnaru
SUM() - motham salary entha
AVG() - average salary
MAX() - highest salary
MIN() - lowest salary

SELECT COUNT(*) FROM employees;
SELECT SUM(salary) FROM employees;
SELECT AVG(salary) FROM employees;
SELECT MAX(salary), MIN(salary) FROM employees;

5. GROUP BY - Group chesi lekka veyadam

GROUP BY = okate department lo entha mandi unnaro telusukodaniki

SELECT department_id, COUNT(*) FROM employees GROUP BY department_id;
SELECT department_id, AVG(salary) FROM employees GROUP BY department_id;
SELECT department_id, SUM(salary) FROM employees GROUP BY department_id;

6. HAVING - Group ni filter cheyadam

WHERE = row ni filter chestundi
HAVING = group ni filter chestundi (GROUP BY tarvata)

SELECT department_id, COUNT(*) as emp_count 
FROM employees GROUP BY department_id 
HAVING COUNT(*) > 5; -- 5 kante ekkuva unna departments matrame

7. LIKE - Similar names vetakadam

LIKE = peru lo okati unda ani vetakadaniki
% = entha letters ayina parvaledu
_ = okka letter

SELECT * FROM employees WHERE first_name LIKE 'J%'; -- J tho start ayye peru
SELECT * FROM employees WHERE first_name LIKE '%a%'; -- lo a unna peru
SELECT * FROM employees WHERE email LIKE '%@sqltutorial.org';
